      Decorator to mark properties of a class for nested serialization. Primitives and enums will be serialized as is. Serializables will be serialized nested (via Serializable.serialize/Serializable.deserialize).

      • To serialize a function type (typeof _type), use the serializeFunction decorator.
      • To serialize Node or SerializableResource references, use the serializeReference decorator.
      • To serialize with type information for polymorphic reconstruction, use the serializeReconstruct decorator.

      Decorated properties are serialized by calling serializeDecorations / deserializeDecorations on an instance. For builtin classes like Component, this is done automatically when the Serializable.serialize / Serializable.deserialize method is called.

      ⚠️ Warning: Do not use with SerializableResource unless you manually deregister them from the project. Otherwise, they will automatically register themselves when deserialized, potentially causing ID conflicts.

      Example:

      import f = FudgeCore;

      export class MyScript extends f.ComponentScript {
        @f.serialize(String) // serialize a string
        public info: string;

        @f.serialize(f.Vector3) // serialize a vector
        public position: f.Vector3 = new f.Vector3(1, 2, 3);

        #size: number = 1;

        @f.serialize(Number) // serialize a number
        public get size(): number {
          return this.#size;
        }

        public set size(_size: number) {
          this.#size = _size;
        }
      }

      Example Nested Resource:

      import f = FudgeCore;

      export class MyScript extends f.ComponentScript {
        @f.serializeReference(f.Material) // serialize a reference to a material in the project
        public material: f.Material;

        @f.serialize(f.Material) // serialize nested
        public nestedMaterial: f.Material;

        public constructor() {
          super();
          this.nestedMaterial = new f.Material("NestedMaterial", f.ShaderPhong);
          
          // ⚠️ important: deregister nested resource, otherwise it will double duty as resource!
          f.Project.deregister(this.nestedMaterial);

          // remove properties that are not needed
          delete this.nestedMaterial.idResource;
          delete this.nestedMaterial.name;
        }
      }
